# นาฉัน — แดชบอร์ด IoT ตรวจสภาพดินในนาข้าว

หน้าเว็บแสดงค่าความชื้นดิน 4 จุด และค่า pH/NPK จากชุดเซ็นเซอร์เคมีดิน 1 จุด มีโหมดข้อมูลจำลอง แผนผังจุดวัด กราฟย้อนหลัง และแจ้งเตือนความชื้นต่ำ รองรับรับข้อมูลจาก Arduino ได้ทั้ง USB Serial และ Wi-Fi บนเครือข่ายเดียวกัน

## เปิดแดชบอร์ดและรับข้อมูลผ่าน Wi-Fi

1. ให้คอมพิวเตอร์ที่ใช้เปิดแดชบอร์ดและบอร์ดอยู่ใน Wi-Fi/เครือข่ายเดียวกัน
2. เปิด PowerShell แล้วรันเซิร์ฟเวอร์ในโฟลเดอร์โปรเจกต์:

   ```powershell
   cd C:\project\rice-field-iot
   python server.py
   ```

3. ถ้า Windows Firewall ถาม ให้เลือกอนุญาต Python เฉพาะ **Private networks** ที่เชื่อถือได้
4. ดู URL ที่ `server.py` แสดงในหน้าต่าง PowerShell:
   - เปิดแดชบอร์ดบนคอมพิวเตอร์: `http://localhost:8000`
   - เปิดแดชบอร์ดบนมือถือ/เครื่องอื่นใน Wi-Fi เดียวกัน: `http://<IP-คอมพิวเตอร์>:8000`
   - URL รับข้อมูลของ Arduino: `http://<IP-คอมพิวเตอร์>:8000/api/telemetry`
5. ตั้งให้บอร์ดเชื่อม Wi-Fi เดียวกัน แล้วส่ง HTTP `POST` เป็น JSON ไปยัง URL รับข้อมูลข้างต้น หน้าแดชบอร์ดจะเปลี่ยนจากข้อมูลตัวอย่างเป็นค่าจากบอร์ดให้อัตโนมัติ

คงหน้าต่าง PowerShell ที่รันเซิร์ฟเวอร์ไว้ตลอดเวลา ถ้า `python` ใช้ไม่ได้ ให้ลอง `py server.py` แทน เบราว์เซอร์ต้องเปิดแดชบอร์ดจากเซิร์ฟเวอร์นี้ (อย่าเปิด `index.html` ด้วยการดับเบิลคลิก) เพราะเซิร์ฟเวอร์นี้เป็นตัวรับข้อมูล Wi-Fi ด้วย

## ตัวอย่างส่งข้อมูลจาก ESP32 ด้วย Arduino IDE

ตัวอย่างนี้แสดงวิธีส่งข้อมูลและยังใช้ค่าจำลอง ให้ใส่ชื่อ Wi-Fi, รหัสผ่าน, IP ของคอมพิวเตอร์ และแทนค่าจำลองด้วยค่าที่อ่าน/ปรับเทียบจากเซ็นเซอร์จริง:

```cpp
#include <WiFi.h>
#include <HTTPClient.h>

const char* WIFI_SSID = "ชื่อ-WiFi";
const char* WIFI_PASSWORD = "รหัสผ่าน-WiFi";
const char* SERVER_URL = "http://192.168.1.50:8000/api/telemetry";

void setup() {
  Serial.begin(115200);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWi-Fi connected");
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(SERVER_URL);
    http.addHeader("Content-Type", "application/json");

    String body =
      "{\"moisture\":[61,55,68,47],\"ph\":6.4,\"n\":38,\"p\":21,\"k\":67}";
    int status = http.POST(body);

    Serial.printf("HTTP status: %d\n", status);
    http.end();
  }

  delay(5000);
}
```

ถ้าใช้บอร์ด Arduino รุ่นอื่นที่มี Wi-Fi ในตัว หรือใช้ Wi-Fi shield/module ให้ใช้ไลบรารี Wi-Fi/HTTP ตามรุ่นบอร์ด แต่ส่ง `POST` body และ URL รูปแบบเดียวกัน:

```json
{"moisture":[61,55,68,47],"ph":6.4,"n":38,"p":21,"k":67}
```

`moisture` ต้องมีตัวเลข 4 ค่า เรียงตามจุด 1–4; `ph` ต้องอยู่ในช่วง 0–14; `n`, `p`, `k` เป็นตัวเลขไม่ติดลบตามหน่วย/การปรับเทียบของโมดูล ส่งข้อมูลเป็นระยะ เช่น ทุก 5–30 วินาที

## ใช้ USB Serial

หากบอร์ดไม่มี Wi-Fi สามารถใช้ Chrome หรือ Edge บนคอมพิวเตอร์ที่ต่อ USB กับบอร์ดได้: ปิด Serial Monitor แล้วกด **เชื่อมต่อ USB** เลือกพอร์ต จากนั้นให้ Arduino ส่ง JSON หนึ่งบรรทัดต่อชุดข้อมูลด้วย baud rate `9600` รูปแบบเดียวกับตัวอย่างด้านบน

## แก้ปัญหาเบื้องต้น

- **หน้าเว็บแสดงข้อมูลตัวอย่างอยู่:** ตรวจว่าเปิดด้วย `python server.py` ไม่ใช่ `python -m http.server` และเปิดหน้าเว็บจาก URL/พอร์ตเดียวกัน จากนั้นตรวจ URL/IP ฝั่งบอร์ด
- **บอร์ดส่งไม่ถึงเซิร์ฟเวอร์:** ตรวจว่า IP คอมพิวเตอร์ถูกต้อง ทั้งสองเครื่องอยู่เครือข่ายเดียวกัน และ Firewall อนุญาต Private network; Wi-Fi สำหรับแขกหรือการตั้งค่า client/AP isolation อาจบล็อกการติดต่อระหว่างอุปกรณ์
- **ใช้มือถือเปิดหน้าเว็บ:** ใช้ `http://<IP-คอมพิวเตอร์>:8000` ไม่ใช่ `localhost` เพราะ `localhost` บนมือถือหมายถึงตัวมือถือเอง
- **ต้องใช้เน็ตนอกบ้าน:** ตัวอย่างนี้ออกแบบสำหรับเครือข่ายภายในเท่านั้น อย่าเปิดพอร์ตเซิร์ฟเวอร์นี้สู่สาธารณะหรือทำ port forwarding หากต้องการใช้งานระยะไกล ควรวางระบบผ่าน VPN หรือ MQTT broker ที่มี TLS และการยืนยันตัวตน

## ข้อควรทราบเกี่ยวกับเซ็นเซอร์

ตัวอย่าง Arduino ข้างต้นใช้ค่าคงที่เพื่อทดสอบการส่งผ่าน Wi-Fi เท่านั้น ไม่ได้อ่านฮาร์ดแวร์ ตัวเซ็นเซอร์จริง, รุ่นบอร์ด, การต่อสาย, หน่วย NPK และวิธีปรับเทียบยังไม่ได้ระบุ หน้าเว็บแสดงค่าตามที่บอร์ดส่งมา ไม่ได้แปลงหน่วยหรือให้คำแนะนำการใส่ปุ๋ย

เปอร์เซ็นต์ความชื้นต้องปรับเทียบค่าดิบกับสภาพแห้ง/เปียกจริง ส่วนค่า pH และ NPK ขึ้นกับรุ่นโมดูลและวิธีวัด ตรวจสอบแรงดันไฟฟ้าและการป้องกันน้ำตามสเปกก่อนต่อเซ็นเซอร์

## บันทึกวันปลูกและประเมินระยะอัตโนมัติ

เปิดเว็บครั้งแรกแล้วบันทึกวันที่ปลูก/ปักดำเพียงครั้งเดียว ระบบเก็บวันดังกล่าวใน `localStorage` ของเบราว์เซอร์ และคำนวณอายุข้าวใหม่อัตโนมัติเมื่อเปิดเว็บหรือเมื่อวันเปลี่ยน ไม่ต้องเลือกช่วงเดิมซ้ำ ภาพประกอบระยะจะแสดงตามอายุโดยประมาณ:

- ต้นอ่อน: 0–19 วัน
- แตกกอ: 20–49 วัน
- ตั้งท้อง: 50–79 วัน
- ออกรวง: 80–99 วัน
- สุกแก่: 100 วันขึ้นไป (เกณฑ์ปลายระยะอ้างอิง 100–120 วัน)

ช่วงอายุที่ผู้ใช้ให้มาซ้อนทับกัน เช่น ตั้งท้อง 50–100 วันกับออกรวง 80–100 วัน ระบบจึงใช้ลำดับข้างต้นเพื่อให้ได้ผลอัตโนมัติเพียงระยะเดียวต่อวัน ช่วงจริงขึ้นกับพันธุ์ วิธีปลูก และสภาพแปลง ไม่ใช่การวิเคราะห์ภาพหรือการตรวจพบระยะจริงด้วยกล้อง ควรสังเกตต้นข้าวประกอบ เมื่อเกิน 120 วันระบบยังแสดงสุกแก่และเตือนให้ตรวจความแก่/ปรึกษาเจ้าหน้าที่

เปลี่ยนวันปลูกได้จากปุ่ม **แก้วันปลูก** หากเริ่มรอบใหม่ วันปลูกถูกเก็บแยกตามเบราว์เซอร์และเครื่องที่เปิดหน้าเว็บ หากใช้มือถืออีกเครื่อง ให้บันทึกวันปลูกในเบราว์เซอร์เครื่องนั้นด้วย ล้าง site data หรือใช้โหมดไม่ระบุตัวตนจะลบวันปลูกที่บันทึกไว้

แนวทางบนหน้าเว็บเป็นข้อมูลทั่วไปเพื่อช่วยสังเกต ไม่ใช่สูตรให้น้ำหรือปุ๋ยเฉพาะพันธุ์ ค่าเตือนความชื้น 35% เป็นเกณฑ์ตัวอย่าง ต้องปรับหลังสอบเทียบหัววัดกับดินและวิธีวัดจริง ค่า pH แสดงคำเตือนเบื้องต้น โดยการแก้ดินกรดต้องทราบความต้องการปูนจากผลวิเคราะห์ดินก่อน ส่วนสถานะ N/P/K ให้เลือกจากรายงานผลตรวจดิน ไม่ได้อนุมานตัวเลขขาด–เกินจากเซ็นเซอร์ เนื่องจากหน่วยและการสอบเทียบของโมดูลแตกต่างกัน อย่ากำหนดอัตราปูนหรือปุ๋ยจากค่าดิบนี้ ให้ยืนยันกับห้องปฏิบัติการหรือหน่วยงานเกษตรในพื้นที่และทำตามฉลากผลิตภัณฑ์
